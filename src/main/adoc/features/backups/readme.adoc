== Troubleshooting

.Sizing backups
[source, bash]
----
$ du -h --max-depth=1 /var/lib/docker/volumes/
----

.Pre-Backup/Restore
[source, bash]
----
# Options
$ export BACKUP_DATE=$(date '+%Y-%m-%d')
$ export BACKUP_FOLDER=/backups/victoriametrics

# Stopping and preventing data ingestion
$ docker stop ti-dhome_victoriametrics_1

# Preparing destination
$ mkdir -p $(WORKSPACE)/backup/victoriametrics
----

.Backup
[source, bash]
----
# Backup: mount volume in other docker instance, and run rsync
$ docker run --rm -it -v ti-dhome_victoria-metrics-data:/victoria-metrics-data -v $(BACKUP_FOLDER):/backup/victoriametrics -w /victoria-metrics-data alpine sh -c "apk add rsync && rsync --delete -rtD --info=progress2 /victoria-metrics-data/* /backup/victoriametrics/"
----

.Post-Backup/Restore
[source, bash]
----
# Restart DB for data ingestion
$ docker start ti-dhome_victoriametrics_1
----
